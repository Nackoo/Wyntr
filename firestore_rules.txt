rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents { 
     
    function isUserJoined(communityId) {
      return request.auth != null && exists(
        /databases/$(database)/documents/communities/$(communityId)/members/$(request.auth.uid)
      );
    } 

		function isBanned() {
      let userPath = /databases/$(database)/documents/users/$(request.auth.uid);
      return request.auth != null &&
             exists(userPath) &&
             ("banned" in get(userPath).data) &&
             get(userPath).data.banned == true;
    }

    function isSuspended() {
			let userPath = /databases/$(database)/documents/users/$(request.auth.uid);
      return request.auth != null &&
             exists(userPath) &&
             ("suspendedUntil" in get(userPath).data) &&
             get(userPath).data.suspendedUntil is timestamp &&
             get(userPath).data.suspendedUntil > request.time &&
 						 ("suspended" in get(userPath).data) &&
             get(userPath).data.suspended == true;
    }

    function isUserAdmin() {
      let userPath = /databases/$(database)/documents/users/$(request.auth.uid);
      return request.auth != null &&
             exists(userPath) &&
             get(userPath).data.role == "admin";
    }

    function isCommunityPrivate(communityId) {
      return get(
        /databases/$(database)/documents/communities/$(communityId)
      ).data.private == true;
    }

    function isCommunityOwner(communityId) {
      let community = get(/databases/$(database)/documents/communities/$(communityId)).data;

      return request.auth != null && community.creatorId == request.auth.uid;
    }
    
    function isCommunityModerator(communityId) {
      let community = get(/databases/$(database)/documents/communities/$(communityId)).data;

      return request.auth != null && (
        community.creatorId == request.auth.uid ||
        request.auth.uid in community.admin
      );
    }

    match /banned/{userId} {
      allow read: if request.auth != null;
      allow create, delete: if request.auth != null && isUserAdmin(); 
      allow update: if false; 
    }

    match /communities/{communityId} {
      allow read: if request.auth != null && !isBanned();
      allow create: if request.auth != null
        && request.resource.data.creatorId == request.auth.uid
        && request.resource.data.name is string
        && request.resource.data.name.size() > 0
        && !isBanned() && !isSuspended();

      allow update: if request.auth != null && (
        // UNSAFE: REMOVE LATER
        request.writeFields.hasOnly(["membersCount", "members"]) ||
        request.writeFields.hasOnly(["posts", "lastActivity"]) ||

        request.writeFields.hasOnly(["admin"]) && isCommunityOwner(communityId) ||
        request.writeFields.hasOnly(["pinned"]) && isCommunityModerator(communityId) ||
        request.writeFields.hasOnly(["name", "lowerCase", "description", "banner", "avatar", "acceptingApplications", "private", "tags", "requirements", "rules"]) && isCommunityModerator(communityId)
      );

      allow delete: if request.auth != null 
        && isCommunityOwner(communityId) || isUserAdmin();
      
      match /members/{userId} {
        allow create, delete: if request.auth != null;
        allow read: if request.auth != null && (
          resource == null || 
          isCommunityModerator(communityId) || 
          resource.data.status != "private"
        );
        allow update: if request.auth != null && ((
          request.auth.uid == userId && 
          request.resource.data.diff(resource.data).affectedKeys().hasOnly(['status']) && 
          request.resource.data.status == 'private') ||
          isCommunityModerator(communityId)
        );
      }
      
      match /bans/{userId} {
        allow create, read, delete: if request.auth != null;
        allow update: if false;
      }
    }
    
    match /communities/{communityId}/posts/{postId} {
      allow read: if request.auth != null && !isBanned() && (
        !isCommunityPrivate(communityId) ||
        isUserJoined(communityId)) ||
        isUserAdmin()
      ;

      allow create: if request.auth != null &&
        request.resource.data.uid == request.auth.uid &&
        request.resource.data.text is string &&
        request.resource.data.text.size() > 0 &&
        request.resource.data.text.size() < 2001 &&
        isUserJoined(communityId) &&
        !isBanned() && !isSuspended();

      allow update: if request.auth != null && (
        request.writeFields.hasOnly(["likeCount"]) ||
        request.writeFields.hasOnly(["poll"]) ||
        request.writeFields.hasOnly(["commentCount"]) ||
        request.writeFields.hasOnly(["retweetCount"]) ||
        request.writeFields.hasOnly(["viewsCount"]) ||
        request.writeFields.hasOnly(["commentCount", "donations"]) ||
        
        (resource.data.uid == request.auth.uid && (
        request.writeFields.hasOnly(["text", "title", "edited", "language", "editAfterComment"]) ||
        request.writeFields.hasOnly(["text"]) ||
        request.writeFields.hasOnly(["muteNotif", "noPrivateReply", "sensitiveMedia"]) ||
        request.writeFields.hasOnly(["connectedWynt", "postedInPublic"]) ||
        request.writeFields.hasOnly(["archived"])))
      );

      allow delete: if request.auth != null && (
        resource.data.uid == request.auth.uid ||
        isUserAdmin() ||
        isCommunityModerator(communityId)
      );

      match /votes/{userId} {
        allow read: if request.auth != null;
        allow create: if request.auth != null && request.auth.uid == userId && !isBanned() && !isSuspended();
        allow delete, update: if false;
      }
      
      match /views/{userId} {
        allow read: if request.auth != null;
        allow delete: if false;
        allow create: if request.auth != null && request.auth.uid == userId;
      }
    }

    function isCommentHideUpdate() {
      return request.writeFields.hasOnly([
        'isHidden',
        'hiddenReason',
        'hiddenByAdmin',
        'hiddenByAuthority'
      ]);
    }

    function isCommunityPostOwner(communityId, postId) {
      return get(
        /databases/$(database)/documents/communities/$(communityId)/posts/$(postId)
      ).data.uid == request.auth.uid;
    }

    function isPostOwnerHideAllowed() {
      return (
        !resource.data.hiddenByAuthority &&
        !resource.data.hiddenByAdmin &&
        !request.resource.data.hiddenByAuthority &&
        !request.resource.data.hiddenByAdmin
      );
    }

    function isPostOwner(postId) {
      return get(
        /databases/$(database)/documents/tweets/$(postId)
      ).data.uid == request.auth.uid;
    }

    function isAuthorityHidden() {
      return resource.data.hiddenByAuthority == true;
    }

    function isAdminHidden() {
      return resource.data.hiddenByAdmin == true;
    }
    
    match /communities/{communityId}/posts/{postId}/comments/{commentId} {
      allow read: if request.auth != null && (
        !isCommunityPrivate(communityId) ||
        isUserJoined(communityId) ||
        isUserAdmin() ||
        !isBanned()
      );

      allow create: if request.auth != null &&
        request.resource.data.uid == request.auth.uid &&
        !isBanned() && !isSuspended();

      allow update: if request.auth != null && (
        resource.data.uid == request.auth.uid && (
          request.writeFields.hasOnly(["text", "edited", "language", "editAfterComment"]) ||
          request.writeFields.hasOnly(["muteNotif", "sensitiveMedia"])
        ) ||

        // UNSAFE: REMOVE LATER
        request.writeFields.hasOnly(['ownerReplied']) ||
        request.writeFields.hasOnly(['ownerReplying']) ||

        request.writeFields.hasOnly(['pinned', 'hasBeenPinned']) ||
        request.writeFields.hasOnly(['replyCount', 'ownerReplied']) ||
        request.writeFields.hasOnly(['likeCount', 'likedByCreator']) ||
        request.writeFields.hasOnly(['likeCount', 'likedByCreator', 'viewsCount']) ||
        request.writeFields.hasOnly(['likeCount']) ||
        request.writeFields.hasOnly(['replyCount']) ||
        request.writeFields.hasOnly(['viewsCount']) ||
        request.writeFields.hasOnly(['retweetCount']) ||
        
        isCommentHideUpdate() && (
          isUserAdmin() ||

          (isCommunityModerator(communityId) &&
          !request.resource.data.hiddenByAuthority) ||

          (isCommunityPostOwner(communityId, postId) && isPostOwnerHideAllowed())
        ) ||

        isUserAdmin()
      );

      allow delete: if request.auth != null && (
        resource.data.uid == request.auth.uid ||
        isUserAdmin() ||
        isCommunityModerator(communityId)
      );
      
      match /votes/{userId} {
        allow read: if request.auth != null;
        allow create: if request.auth != null && request.auth.uid == userId && !isBanned() && !isSuspended();
        allow delete, update: if false;
      }
      
      match /views/{userId} {
        allow read: if request.auth != null;
        allow create: if request.auth != null && request.auth.uid == userId;
        allow delete: if false
      }
    }
    
    match /susList/{userId} {
      allow create, read, update: if request.auth != null && isUserAdmin();
      allow delete: if false; 
    }

    match /tags/{tagId} {
      allow read: if request.auth != null;
      allow create, update: if request.auth != null;
    }

    match /users/{userId} {
      allow get, list: if request.auth != null;

      allow create: if request.auth != null &&
                    request.auth.uid == userId &&
                    !("role" in request.resource.data);

      allow update: if request.auth != null && (
          (request.writeFields.hasOnly(["posts"]) && (request.auth.uid == userId || isUserAdmin())) ||
          (request.writeFields.hasOnly(["balance"]) &&
          request.resource.data.balance is int &&
          request.resource.data.balance > resource.data.balance) ||
          (request.writeFields.hasOnly(["followers"])) ||
          request.writeFields.hasOnly(["communitiesCount"]) ||
          (isUserAdmin() && (
            request.writeFields.hasOnly(["banned"]) ||
            request.writeFields.hasOnly(["banned", "bannedFor"]) ||
            request.writeFields.hasOnly(["suspended", "suspendedFor", "suspendedUntil"]) ||
            request.writeFields.hasOnly(["suspended"]))
          ) ||
          !("role" in request.resource.data.diff(resource.data).affectedKeys())
        );

      allow delete: if false;
        
      match /mentioned/{tweetId} {
        allow read: if request.auth != null;
        allow create: if request.auth != null &&
          exists(/databases/$(database)/documents/tweets/$(tweetId)) &&
          userId in get(/databases/$(database)/documents/tweets/$(tweetId)).data.mentions;
        allow delete: if request.auth != null &&
          exists(/databases/$(database)/documents/tweets/$(tweetId)) &&
          (request.auth.uid == get(/databases/$(database)/documents/tweets/$(tweetId)).data.uid || isUserAdmin()
          );
      }
      
      match /blocks/{blockedUserId} {
        allow read: if request.auth != null;
        allow write: if request.auth != null && request.auth.uid == userId;
      }
      
      match /posts/{postId} {
        allow read: if request.auth != null;
        allow write: if request.auth != null && (
          request.auth.uid == userId ||
          isUserAdmin()
        );
      }
      
      match /notifications/{notificationId} {
        allow create, read: if request.auth != null && !isBanned();

        allow delete: if request.auth != null
          && request.auth.uid == userId
          && !(
            resource.data.type in [
              "community-delete",
              "comment-delete",
              "tweet"
            ]
          );

        allow update: if request.auth != null
        	&& request.resource.data.type == resource.data.type;
      }

      match /bookmarks/{folderId} {
        allow read, write: if request.auth != null && request.auth.uid == userId;

        match /items/{tweetId} {
          allow read, write: if request.auth != null && request.auth.uid == userId;
        }
      }
      
      match /highlights/{folderId} {
        allow read: if request.auth != null;
        allow write: if request.auth != null && request.auth.uid == userId;
       
        match /items/{tweetId} {
          allow read: if request.auth != null;
          allow write: if request.auth != null && request.auth.uid == userId;
        }
      }
      
      match /following/{targetId} {
        allow read: if request.auth != null &&
          ((resource == null || request.auth.uid == userId || request.auth.uid == targetId) ||
          (
            get(/databases/$(database)/documents/users/$(userId)).data.get('cannotSeeFollows', false) != true &&
            resource.data.status != "private"
          ));

        allow create: if request.auth != null && request.auth.uid == targetId && !isBanned() && !isSuspended();

        allow update: if request.auth != null && !isBanned() && !isSuspended() && (
          (request.auth.uid == targetId || request.auth.uid == userId) && request.resource.data.status == "private"
        );

        allow delete: if request.auth != null && request.auth.uid == userId && !isBanned() && !isSuspended();
      }

      match /followers/{followerId} {
        allow read: if request.auth != null &&
          ((resource == null || request.auth.uid == userId || request.auth.uid == followerId) ||
          (
            get(/databases/$(database)/documents/users/$(userId)).data.get('cannotSeeFollowers', false) != true &&
            resource.data.status != "private"
          ));

        allow create: if request.auth != null && request.auth.uid == followerId && !isBanned() && !isSuspended();

        allow update: if request.auth != null && !isBanned() && !isSuspended() && (
          (request.auth.uid == followerId || request.auth.uid == userId) && request.resource.data.status == "private"
        );

        allow delete: if request.auth != null && request.auth.uid == followerId && !isBanned() && !isSuspended();
      }
    }

    match /tweets/{tweetId} {
      allow read: if request.auth != null && !isBanned();

      allow create: if request.auth != null &&
        request.resource.data.uid == request.auth.uid
        && !isBanned() && !isSuspended();

      allow update: if request.auth != null && (
        (request.writeFields.size() == 1 &&
         request.writeFields.hasOnly(['likeCount'])) ||
        request.writeFields.hasOnly(["poll"]) ||
        (request.writeFields.hasOnly(['donations'])) ||
        (request.writeFields.size() == 1 &&
        (request.writeFields.hasOnly(['commentCount']) ||
        request.writeFields.hasOnly(['retweetCount']))) ||
        (request.writeFields.size() == 1 &&
        request.writeFields.hasOnly(['viewsCount']) &&
        request.resource.data.viewsCount == resource.data.viewsCount + 1) ||

        (resource.data.uid == request.auth.uid && (
        request.writeFields.hasOnly(["text", "title", "edited", "language", "editAfterComment"]) ||
        request.writeFields.hasOnly(["muteNotif", "noPrivateReply", "sensitiveMedia"]) ||
        request.writeFields.hasOnly(["connectedWynt", "postedInPublic"])) ||
        request.writeFields.hasOnly(["archived"]))
      );

      allow delete: if request.auth != null && (
        resource.data.uid == request.auth.uid ||
        isUserAdmin()
      );
      
      match /votes/{userId} {
        allow read: if request.auth != null;
        allow create: if request.auth != null && request.auth.uid == userId && !isBanned() && !isSuspended();
        allow delete, update: if false;
      }

      match /views/{userId} {
        allow read: if request.auth != null;
        allow create: if request.auth != null && request.auth.uid == userId;
        allow delete: if false
      }

      match /comments/{commentId} {
        allow read: if request.auth != null && !isBanned();

        allow create: if request.auth != null &&
          request.resource.data.uid == request.auth.uid &&
          !isBanned() && !isSuspended();

        allow update: if request.auth != null && (
          (request.writeFields.size() == 1 &&
            request.writeFields.hasOnly(['likeCount']) ||
            request.writeFields.hasOnly(['viewsCount']) ||
            request.writeFields.hasOnly(['replyCount']) ||
            request.writeFields.hasOnly(['retweetCount'])
          ) ||

          request.writeFields.hasOnly(['likeCount', 'likedByCreator']) ||
          request.writeFields.hasOnly(['likeCount', 'likedByCreator', 'viewsCount']) ||

          (isUserAdmin() && (
            request.writeFields.hasOnly(['isHidden', 'hiddenByAuthority', 'hiddenReason']) ||
            request.writeFields.hasOnly(['isHidden', 'hiddenByAdmin', 'hiddenReason', 'hiddenByAuthority'])
          )) ||

          (resource.data.uid == request.auth.uid && 
            request.writeFields.hasOnly(["text", "edited", "language", "editAfterComment"]) ||
            request.writeFields.hasOnly(["muteNotif", "sensitiveMedia"]) ||
            (request.writeFields.size() == 1 && request.writeFields.hasOnly(['text']))
          ) ||

          request.writeFields.hasOnly(['ownerReplied']) ||
          request.writeFields.hasOnly(['ownerReplying']) ||
          request.writeFields.hasOnly(["pinned", "hasBeenPinned"]) ||
          request.writeFields.hasOnly(["ownerReplied", "replyCount"]) ||

          isCommentHideUpdate() && (
            isUserAdmin() ||
            (isPostOwner(tweetId) && isPostOwnerHideAllowed())
          ) ||

          isUserAdmin()
        );      
        
        allow delete: if request.auth != null && (
          request.auth.uid == resource.data.uid ||
          isUserAdmin()
        );
        
        match /votes/{userId} {
          allow read: if request.auth != null;
          allow create: if request.auth != null && request.auth.uid == userId && !isBanned() && !isSuspended();
          allow delete, update: if false;
        }
        
        match /views/{userId} {
          allow read: if request.auth != null;
          allow create: if request.auth != null && request.auth.uid == userId;
          allow delete: if false
        }
      }
    }

    function isParentOwner(parentData) {
      return request.auth != null && parentData.uid == request.auth.uid;
    }

    match /communities/{communityId}/posts/{postId}/likes/{userId} {
      allow read: if request.auth != null && (
        resource == null || 
        request.auth.uid == userId ||
        isParentOwner(get(/databases/$(database)/documents/communities/$(communityId)/posts/$(postId)).data) || 
        resource.data.status != "private"
      );
      allow create, delete: if request.auth.uid == userId && !isBanned() && !isSuspended();
      allow update: if request.auth != null && 
                       request.auth.uid == userId && 
                       request.writeFields.hasOnly(['status']) && 
                       request.resource.data.status == 'private';
    }

    match /communities/{communityId}/posts/{postId}/comments/{commentId}/likes/{userId} {
      allow read: if request.auth != null && (
        resource == null || 
        request.auth.uid == userId || 
        isParentOwner(get(/databases/$(database)/documents/communities/$(communityId)/posts/$(postId)/comments/$(commentId)).data) || 
        resource.data.status != "private"
      );
      allow create, delete: if request.auth.uid == userId && !isBanned() && !isSuspended();
      allow update: if request.auth != null && 
                       request.auth.uid == userId && 
                       request.writeFields.hasOnly(['status']) && 
                       request.resource.data.status == 'private';
    }

    match /tweets/{tweetId}/likes/{userId} {
      allow read: if request.auth != null && (
        resource == null || 
        request.auth.uid == userId ||
        isParentOwner(get(/databases/$(database)/documents/tweets/$(tweetId)).data) || 
        resource.data.status != "private"
      );
      allow create, delete: if request.auth.uid == userId && !isBanned() && !isSuspended();
      allow update: if request.auth != null && 
                       request.auth.uid == userId && 
                       request.writeFields.hasOnly(['status']) && 
                       request.resource.data.status == 'private';
    }

    match /tweets/{tweetId}/comments/{commentId}/likes/{userId} {
      allow read: if request.auth != null && (
        resource == null || 
        request.auth.uid == userId || 
        isParentOwner(get(/databases/$(database)/documents/tweets/$(tweetId)/comments/$(commentId)).data) || 
        resource.data.status != "private"
      );
      allow create, delete: if request.auth.uid == userId && !isBanned() && !isSuspended();
      allow update: if request.auth != null && 
                       request.auth.uid == userId && 
                       request.writeFields.hasOnly(['status']) && 
                       request.resource.data.status == 'private';
    }
  }
}
